# Nerves U-Boot environment for STM32MP157F-DK2
#
# U-Boot loads this file at boot. It tells U-Boot where to find
# the kernel, device tree, and root filesystem.
#
# Load addresses are specific to the STM32MP1 memory map:
#   DDR starts at 0xC0000000 (512MB)
#   We place kernel, DTB, and ramdisk at safe offsets in DDR

bootpart=0:4
bootdir=/boot
bootfile=zImage
fdtfile=st/stm32mp157c-dk2.dtb

nerves_fw_active=a

# Load addresses in DDR
kernel_addr_r=0xc2000000
fdt_addr_r=0xc4000000
ramdisk_addr_r=0xc4400000

# Kernel command line
# console=ttySTM0 — UART4 via ST-LINK virtual COM port
# root=/dev/mmcblk0p4 — rootfs-a partition
# rootfstype=squashfs — read-only compressed filesystem
bootargs=console=ttySTM0,115200 root=/dev/mmcblk0p4 rootfstype=squashfs rootwait

# Boot command: load kernel + DTB from microSD, boot
bootcmd=load mmc ${bootpart} ${kernel_addr_r} ${bootdir}/${bootfile}; \
  load mmc ${bootpart} ${fdt_addr_r} ${bootdir}/${fdtfile}; \
  bootz ${kernel_addr_r} - ${fdt_addr_r}
